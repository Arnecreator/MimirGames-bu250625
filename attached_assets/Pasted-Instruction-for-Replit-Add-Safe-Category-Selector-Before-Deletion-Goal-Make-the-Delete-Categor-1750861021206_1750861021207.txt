Instruction for Replit – Add Safe Category Selector Before Deletion
🎯 Goal
Make the “Delete Category” action safer and more transparent by showing a category selection modal before any deletion happens.

🧩 What to do
1. Replace current onclick="confirmDeleteCategory()" with this:
html
Copy
Edit
<button class="admin-btn btn-danger" onclick="showDeleteCategoryModal()">Delete Category</button>
2. Add this function to your JavaScript:
js
Copy
Edit
function showDeleteCategoryModal() {
  const categories = [
    "History", "Science", "Sports", "Geography", "Literature", "Math",
    "Movies", "Music", "Art", "Technology", "Food", "Misc"
  ];

  let optionsHtml = categories.map(cat =>
    `<option value="${cat}">${cat}</option>`
  ).join('');

  Swal.fire({
    title: 'Select category to delete',
    html: `
      <select id="category-to-delete" class="swal2-input">
        ${optionsHtml}
      </select>
    `,
    showCancelButton: true,
    confirmButtonText: 'Delete',
    confirmButtonColor: '#d33',
    cancelButtonText: 'Cancel',
    preConfirm: () => {
      const selected = document.getElementById('category-to-delete').value;
      return selected;
    }
  }).then((result) => {
    if (result.isConfirmed && result.value) {
      deleteCategory(result.value);
    }
  });
}
3. Create or modify the actual deletion function:
js
Copy
Edit
function deleteCategory(category) {
  fetch(`/api/deleteCategory?category=${category}`, {
    method: 'DELETE'
  })
    .then(res => res.json())
    .then(data => {
      if (data.success) {
        Swal.fire({
          icon: 'success',
          title: `Category "${category}" deleted.`,
          timer: 1500,
          toast: true,
          position: 'top-end'
        });
      } else {
        throw new Error(data.message);
      }
    })
    .catch(err => {
      Swal.fire({
        icon: 'error',
        title: 'Failed to delete category',
        text: err.message
      });
    });
}

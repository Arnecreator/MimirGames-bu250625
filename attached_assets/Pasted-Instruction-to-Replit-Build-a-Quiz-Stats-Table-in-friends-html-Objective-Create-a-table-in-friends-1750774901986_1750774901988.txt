✅ 2. Add this JavaScript to the bottom of the file (before </body>)
This code dynamically populates the table using data from the friends array (to be replaced later by API calls):

<script>
  // Placeholder data – replace with fetch call to get real friends
  const friends = [
    {
      username: "alice2024",
      quizPlayedDays: 23,
      quizAnswers: 184,
      quizCorrect: 156,
      quizPercentCorrect: 85,
      quizFullScoreGames: 8,
      quizStreak: 8,
      regDate: "240203"
    },
    {
      username: "bob456",
      quizPlayedDays: 26,
      quizAnswers: 248,
      quizCorrect: 189,
      quizPercentCorrect: 76,
      quizFullScoreGames: 15,
      quizStreak: 7,
      regDate: "240128"
    }
  ];

  function calculateFullScorePercent(user) {
    if (!user.quizPlayedDays || user.quizPlayedDays === 0) return 0;
    return ((user.quizFullScoreGames / user.quizPlayedDays) * 100).toFixed(0);
  }

  function renderQuizTable() {
    const tableBody = document.getElementById("quizFriendsTable");
    tableBody.innerHTML = "";
    friends.forEach(user => {
      const row = document.createElement("tr");
      row.innerHTML = `
        <td>${user.username}</td>
        <td>${user.quizPlayedDays}</td>
        <td>${user.quizAnswers}</td>
        <td>${user.quizCorrect}</td>
        <td>${user.quizPercentCorrect}%</td>
        <td>${user.quizFullScoreGames}</td>
        <td>${calculateFullScorePercent(user)}%</td>
        <td>${user.quizStreak}</td>
        <td>${user.regDate}</td>
        <td><button onclick="removeFriend('${user.username}')">🗑️</button></td>
      `;
      tableBody.appendChild(row);
    });
  }

  function removeFriend(username) {
    alert("Removing: " + username);
    // Later: Replace with fetch to remove friend in backend
  }

  renderQuizTable();
</script>
🧩 Notes:
friends is placeholder data. Replace it later with a fetch() call to your API that returns the list of friend usernames and their quiz stats.

removeFriend(username) should eventually call a PATCH/DELETE endpoint to update the user's friend list in MongoDB.

Column headers (Qpd, #answ, etc.) are intentionally aligned with the admin panel.

document.getElementById('add-all-btn').addEventListener('click', async () => {
  const categories = [
    "History", "Science", "Sports", "Geography",
    "Literature", "Math", "Movies", "Music",
    "Art", "Technology", "Food", "Misc"
  ];

  for (const category of categories) {
    try {
      const res = await fetch(`/api/addQuestions?category=${category}`);
      const data = await res.json();

      if (data.success) {
        // Find the corresponding button
        const btn = document.querySelector(`.add-btn[data-category="${category}"]`);
        if (btn) {
          let count = parseInt(btn.getAttribute('data-count')) || 0;
          count += 10;
          btn.setAttribute('data-count', count);
          btn.textContent = `Add to ${category} (${count})`;
        }
      } else {
        console.error(`Error adding to ${category}:`, data.message);
      }
    } catch (err) {
      console.error(`Network error for ${category}:`, err);
    }
  }

  Swal.fire({
    icon: 'success',
    title: '✅ Added 10 questions to all categories!',
    showConfirmButton: false,
    timer: 2000,
    toast: true,
    position: 'top-end',
    background: '#a0ffbf'
  });
});
✅ Final Notes
This version uses your real categories – no bulk/unknown insertions.

All buttons update immediately with their new question counts.

Error handling is per-category in the console, but the user only sees a popup if the entire operation finishes.

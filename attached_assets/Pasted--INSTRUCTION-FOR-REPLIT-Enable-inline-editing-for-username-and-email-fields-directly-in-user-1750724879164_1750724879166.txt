Replace popup modal editing with inline editing for `username` and `email` in `public/users.html`.

### 🔧 FRONTEND CHANGES (users.html)

**1. Add these helper functions in a <script> tag in `users.html`:**

```html
<script>
  function makeCellEditable(td, userId, field, originalValue) {
    const input = document.createElement("input");
    input.type = "text";
    input.value = originalValue;
    input.classList.add("inline-input");
    
    td.textContent = "";
    td.appendChild(input);
    input.focus();

    input.addEventListener("keydown", async (e) => {
      if (e.key === "Enter") {
        const newValue = input.value;
        try {
          const res = await fetch(`/api/users/${userId}`, {
            method: "PATCH",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ [field]: newValue })
          });
          if (!res.ok) throw new Error("Update failed");
          td.textContent = newValue;
        } catch (err) {
          td.textContent = originalValue;
          alert("Error updating: " + err.message);
        }
      } else if (e.key === "Escape") {
        td.textContent = originalValue;
      }
    });
  }
</script>
```

**2. Modify the table rendering code in JavaScript to add event listeners:**

When generating each row, apply this after appending the `username` and `email` cells:

```js
usernameCell.addEventListener("dblclick", () => makeCellEditable(usernameCell, user._id, "username", user.username));
emailCell.addEventListener("dblclick", () => makeCellEditable(emailCell, user._id, "email", user.email));
```

### 🖥 BACKEND CHANGES (routes/users.js)

**4. Add PATCH route to update user:**

```js
router.patch("/:id", async (req, res) => {
  try {
    const { id } = req.params;
    const { username, email } = req.body;
    const update = {};
    if (username !== undefined) update.username = username;
    if (email !== undefined) update.email = email;

    const updated = await User.findByIdAndUpdate(id, update, { new: true });
    if (!updated) return res.status(404).send("User not found");
    res.json(updated);
  } catch (err) {
    res.status(500).send("Server error: " + err.message);
  }
});
```

async function fetchFriendsFromDB() {
  try {
    const res = await fetch('/api/friends');
    const data = await res.json();
    return Array.isArray(data.friends) ? data.friends : [];
  } catch (err) {
    console.error("Failed to load friends from DB:", err);
    return [];
  }
}
3. 🔍 For each friend username, fetch full user stats:
javascript
Copy
Edit
async function fetchUserStats(username) {
  try {
    const res = await fetch(`/api/users/${username}`);
    return await res.json();
  } catch (err) {
    console.error(`Failed to fetch user: ${username}`, err);
    return null;
  }
}
4. 📊 Combine dummy data + live DB data:
javascript
Copy
Edit
async function renderQuizTable() {
  const tableBody = document.getElementById("quizFriendsTable");
  tableBody.innerHTML = "";

  // Step 1: Add dummy data first
  friends.forEach(user => appendUserRow(user, tableBody));

  // Step 2: Add real data from DB
  const friendUsernames = await fetchFriendsFromDB();
  for (const username of friendUsernames) {
    const user = await fetchUserStats(username);
    if (user) appendUserRow(user, tableBody);
  }
}

function appendUserRow(user, tableBody) {
  const fullScorePercent = user.quizPlayedDays > 0
    ? ((user.quizFullScoreGames / user.quizPlayedDays) * 100).toFixed(0)
    : 0;

  const row = document.createElement("tr");
  row.innerHTML = `
    <td>${user.username}</td>
    <td>${user.quizPlayedDays}</td>
    <td>${user.quizAnswers}</td>
    <td>${user.quizCorrect}</td>
    <td>${user.quizPercentCorrect}%</td>
    <td>${user.quizFullScoreGames}</td>
    <td>${fullScorePercent}%</td>
    <td>${user.quizStreak}</td>
    <td>${user.regDate}</td>
    <td><button onclick="removeFriend('${user.username}')">🗑️</button></td>
  `;
  tableBody.appendChild(row);
}
